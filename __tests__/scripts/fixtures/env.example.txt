# Reading this file:
#   NAME=value     set; the value works for local development
#   # NAME=value   optional; the value shown is the default unless noted
#   # NAME=<...>   no default; replace the <placeholder> to use the setting
#
# Rules:
#   - Leave an optional variable commented out rather than blank (`NAME=`).
#   - Flags are exactly `true` or `false`, lower case. Lists are comma-separated.
#   - Quote values containing spaces or `<>`. Next.js expands `$NAME` inside
#     values, so write a literal `$` in a secret as `\$`.
#   - Production values belong in the host's secret store, not in a file.

# 4. Mail (Sign-In, Invitations, Bookings)
# ------------------------------------------------------------------------------
# none | smtp | resend
#   none    sends nothing: sign-in mail fails loudly, booking mail is skipped
#           with a warning. Development only; refused in production.
#   smtp    any SMTP server. Locally, MailHog listens on localhost:1025.
#   resend  Resend's HTTP API. Production only; refused elsewhere.
MAIL_PROVIDER=none

# Sender of every mail. Required unless MAIL_PROVIDER=none. Keep the quotes.
# MAIL_FROM="Civo <no-reply@example.org>"

# SMTP (MAIL_PROVIDER=smtp). The host is required.
# MAIL_SMTP_HOST=localhost
# MAIL_SMTP_PORT=1025
# `true` connects over TLS from the first byte (typically port 465).
# MAIL_SMTP_SECURE=false
# Credentials of an authenticated relay: set both or neither. MailHog and
# similar local catchers need none.
# MAIL_SMTP_USER=<username>
# MAIL_SMTP_PASSWORD=<password>

# Resend (MAIL_PROVIDER=resend). The key is required.
# MAIL_API_KEY=<re_...>
# MAIL_API_URL=https://api.resend.com/emails

# ------------------------------------------------------------------------------
# 5. Rate Limiting (Sign-In, Booking, Editor and Release Actions)
# ------------------------------------------------------------------------------
# Counters live in Redis, reached over HTTP (Upstash): no driver, no pool.
# upstash | memory. Default: upstash in production, memory everywhere else.
# `memory` counts per process, so it only suits development and tests;
# production refuses it.
# RATE_LIMIT_BACKEND=memory

# Required in production (and with RATE_LIMIT_BACKEND=upstash). Create a Redis
# database at https://console.upstash.com and copy its REST URL and token; the
# Vercel Upstash integration sets both. Locally, `docker compose up` runs a
# compatible endpoint on http://localhost:8079 (token: see `SRH_TOKEN` in `.env.docker`).
# UPSTASH_REDIS_REST_URL=https://<your-database>.upstash.io
# UPSTASH_REDIS_REST_TOKEN=<token>

# Keys what is counted (addresses, e-mails) so Redis holds none of them in
# clear. At least 32 characters; falls back to AUTH_SECRET.
# RATE_LIMIT_SECRET=<32+ random characters>

# Where a visitor's address comes from: vercel | proxies | none.
# No fixed default: `vercel` when running on Vercel (it sets VERCEL itself),
# `proxies` when AUTH_TRUSTED_PROXIES is set, otherwise `none` (limits that
# count addresses are then skipped and the log says so).
# RATE_LIMIT_CLIENT_ADDRESS_SOURCE=proxies

# Upper bound for one limiter check, in milliseconds (100-10000).
# RATE_LIMIT_TIMEOUT_MS=1500
